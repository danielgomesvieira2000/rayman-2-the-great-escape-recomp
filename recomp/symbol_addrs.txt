// Symbols recovered in phase 01+, and hand-verified names.
//
// splat reads this file and uses these names instead of its generated
// func_<addr> / D_<addr> ones. That matters beyond readability: N64Recomp
// carries a list of known libultra functions and marks any it recognises by
// name as `reimplemented`, so the recompiler skips it and the runtime's native
// version (librecomp / ultramodern) is used instead. That is the only correct
// way to handle routines that touch privileged CPU state, which cannot be
// recompiled at all.
//
// ONLY put a name here when the function's body proves it. A wrong name is
// worse than no name: it silently redirects calls to a runtime function with
// different semantics, and nothing will flag it. Where a body did not prove a
// specific libultra identity, the function is left as func_<addr> and stubbed
// in rayman2.us.toml instead of being given a guessed name.

// ---------------------------------------------------------------------------
// Cache maintenance. Identified from the MIPS cache-op encodings, where the low
// two bits select the cache (0 = I, 1 = D) and bits 4:2 the operation, together
// with the size threshold each routine compares against: 0x4000 is the VR4300's
// 16 KB instruction cache, 0x2000 its 8 KB data cache.
// ---------------------------------------------------------------------------

// Threshold 0x4000; ops 0x00 Index_Invalidate_I and 0x10 Hit_Invalidate_I.
// The only icache routine that takes a range.
osInvalICache = 0x8000CD40; // type:func

// Threshold 0x2000; main loop 0x11 Hit_Invalidate_D, with 0x15
// Hit_Writeback_Invalidate_D on the unaligned head and tail lines so that data
// sharing those lines is not discarded. That head/tail treatment is precisely
// how osInvalDCache is written.
osInvalDCache = 0x80003C30; // type:func

// Threshold 0x2000; op 0x19 Hit_Writeback_D -- writeback without invalidate.
osWritebackDCache = 0x8000CED0; // type:func

// Walks the whole 8 KB data cache from 0x80000000 with 0x01
// Index_Writeback_Invalidate_D and takes no range argument.
osWritebackDCacheAll = 0x80003D90; // type:func

// ---------------------------------------------------------------------------
// Interrupt and status-register access, identified from cop0 register numbers.
// Register 12 is Status, register 9 is Count.
// ---------------------------------------------------------------------------

// Clears Status bit 0 (IE) and returns the previous value of that bit.
__osDisableInt = 0x8000CDC0; // type:func

// ORs its argument back into Status.
__osRestoreInt = 0x8000CDE0; // type:func

// Body is `mfc0 $v0, $9` then `jr $ra`. Cop0 register 9 is Count, the cycle
// counter; a function that only reads it and returns is osGetCount.
osGetCount = 0x8000CD20; // type:func

// `mfc0 $v0, $12` then `jr $ra` -- reads Status and returns it.
// NOT in N64Recomp's reimplemented set, so it is stubbed in the config.
__osGetSR = 0x8000CD30; // type:func

// `mtc0 $a0, $12` then `jr $ra` -- writes Status.
// NOT in N64Recomp's reimplemented set, so it is stubbed in the config.
__osSetSR = 0x8000CEC0; // type:func

// `cfc1 $v0, $31` / `ctc1 $a0, $31` / `jr $ra`: reads the FPU control/status
// register into the return value, then writes the argument into it.
// Return-old-and-set is osSetFpcCsr's contract.
osSetFpcCsr = 0x8000CEB0; // type:func

// ---------------------------------------------------------------------------
// Corrected function boundaries.
//
// splat recovers boundaries from control flow, which misreads large handwritten
// assembly routines whose branches do not look like ordinary intra-function
// flow. Where it splits one function into two, the second half branches
// backwards into the first, and N64Recomp rejects it: "branching outside of the
// function". The fix is to declare the true extent so splat stops splitting.
// ---------------------------------------------------------------------------

// splat ended this at 0x8000C944 (size 0x574) and started a new function there,
// but that "function" branches back to 0x8000C8FC and 0x8000C914, both inside
// this one. The two halves are one routine: 0x574 + 0xF0 = 0x664. It saves and
// restores FPU control state (cfc1/ctc1 $31) and touches cop0, which is the
// shape of a libultra exception handler -- but the body does not prove which
// one, so it keeps its generated name.
func_8000C3D0 = 0x8000C3D0; // type:func size:0x664

// `mtc0 $a0, $11` then `jr $ra`. Cop0 register 11 is Compare, the timer-match
// register; a function that only writes it is the set-Compare routine. This
// name is in N64Recomp's built-in ignored set, so naming it is what makes the
// recompiler drop it.
__osSetCompare = 0x8000CEA0; // type:func

// ---------------------------------------------------------------------------
// The entry stub, and the function it jumps into.
//
// splat sizes func_80000400 at 0xC0, which swallows a whole second function.
// The stub itself is 0x50 bytes: it clears bss, sets sp, and ends with
// `jr $t2` where t2 holds 0x80000450. That jump is INDIRECT, so splat saw no
// call to 0x80000450 and started no function there, and librecomp cannot
// resolve it at run time -- "Failed to find function at 0x80000450", which is
// the first thing the game does and so the first thing that breaks.
//
// This was tried once during phase 03 and appeared to make things worse. That
// measurement was taken through the entrypoint sign-extension bug, which
// crashed the boot DMA before any of this was reached, so it said nothing
// about these boundaries. Re-tested now that the boot path is sound.
// ---------------------------------------------------------------------------
func_80000400 = 0x80000400; // type:func size:0x50
func_80000450 = 0x80000450; // type:func

// ---------------------------------------------------------------------------
// Serial Interface (controller / PIF) access.
//
// The game pokes SI hardware registers directly rather than only through the
// libultra wrappers, and those addresses are KSEG1 (0xA4800018 = SI_STATUS).
// MEM_W maps such an address to rdram + 0x24800018, which is past the 512 MB
// librecomp maps read-write and inside the guard region it deliberately leaves
// PAGE_NOACCESS -- so the read faults rather than returning nonsense, which is
// the guard doing its job.
// ---------------------------------------------------------------------------

// Body is: load 0xA4800018 (SI_STATUS), read it, mask bits 0 and 1
// (DMA_BUSY | IO_READ_BUSY), return whether either is set. That is
// __osSiDeviceBusy and nothing else. Size 0x18 matches its six instructions.
// The name is in N64Recomp's built-in ignored set, so naming it makes the
// recompiler drop it -- and obliges the port to define it (src/port_runtime.cpp).
__osSiDeviceBusy = 0x800115A0; // type:func

// libultra's osInitialize, and the reason the raw SI path is a dead end.
//
// Body, in order: read cop0 Status and OR in 0x20000000 (CU1 -- enable the
// FPU), set the FPU control register to 0x1000800, then spin on
// __osSiRawReadIo(0x1FC007FC) / __osSiRawWriteIo(value | 8) -- the PIF RAM
// handshake that tells the PIF the boot process is finished -- and finally
// take the address of the exception handler to install it. That sequence is
// osInitialize and nothing else.
//
// This name matters more than most. osInitialize is REIMPLEMENTED by librecomp
// (ultra_translation.cpp), so naming it makes the recompiler drop the game's
// copy and run the runtime's instead. That removes the entire raw-hardware
// path underneath it: no PIF RAM access, no SI status polling, no exception
// vector to install. Emulating PIF RAM to satisfy the original would have been
// the wrong fix -- the runtime already owns all of it.
osInitialize = 0x80003EE0; // type:func

// Audio Interface. Called straight from main with a0 = 22050, and the body is
// the frequency-to-DAC-rate conversion in floating point (mtc1 / cvt.d.w on the
// argument) followed by writes to AI_DACRATE (0xA4500010) and AI_BITRATE. That
// is osAiSetFrequency.
//
// Reimplemented by librecomp, which is the point: naming the public libultra
// entry points hands whole subsystems to the runtime, and the low-level
// register pokes beneath them are then never reached. The alternative -- naming
// __osAiDeviceBusy and friends, which are in the recompiler's *ignored* set --
// would oblige this port to emulate the AI registers itself, which is work the
// runtime has already done.
osAiSetFrequency = 0x80007850; // type:func

// ---------------------------------------------------------------------------
// Threading. This is why the screen was black.
//
// The counters in src/rt64_context.cpp showed the VI ticking at 60/s with zero
// display lists, and the entrypoint wrapper showed recomp_entrypoint RETURNING
// within a second. Both are correct behaviour: libultra's boot main calls
// osInitialize, sets the audio rate, creates a thread whose entry is
// 0x800004C0, starts it, and returns. The game then lives on that thread.
//
// It never ran, because the game's own osCreateThread/osStartThread were being
// recompiled. They build an OSThread and push it onto the game's idea of a run
// queue -- but ultramodern owns the scheduler in this port and knows nothing
// about that structure, so the thread is created, marked runnable, and never
// scheduled. Nothing crashes; the game simply stops having anything to do.
//
// Both are reimplemented by librecomp, so naming them hands threading to the
// runtime, which is the only place it can work.
// ---------------------------------------------------------------------------

// Six arguments (thread, id, entry, arg, sp, pri), seeds the saved FPU control
// word with 0x1000800, and points the thread's return address at a cleanup
// routine. Called from main with entry 0x800004C0.
osCreateThread = 0x80003DF0; // type:func

// Takes a thread pointer, enters a critical section via __osDisableInt, reads
// the state field at offset 0x10, accepts STOPPED (1) or WAITING (8), writes
// RUNNABLE (2) and enqueues on the run queue.
osStartThread = 0x800042A0; // type:func

// `if (t == NULL) t = *__osRunningThread; return t->priority;` -- the priority
// field is at offset 0x4, which the osStartThread body above corroborates by
// reading `state` at 0x10 in the same structure. That is osGetThreadPri.
//
// It faulted reading null+4, and the reason generalises: the global at
// D_8001A340 is the game's own __osRunningThread, and nothing sets it any more.
// ultramodern owns the scheduler now, so every piece of the game's libultra
// that reads scheduler state -- the run queue, the running thread -- is reading
// a structure the runtime never touches. Those functions cannot be recompiled;
// they have to be the runtime's. Expect more of these, and expect them to look
// like null dereferences a few fields into a libultra structure.
osGetThreadPri = 0x8000D080; // type:func

// Two arguments (thread, priority), enters a critical section via
// __osDisableInt, then null-defaults the thread to __osRunningThread exactly as
// osGetThreadPri above does. That is osSetThreadPri, and it faulted on the same
// null+4 read for the same reason.
//
// Scale note for whoever continues this: D_8001A340 (__osRunningThread) is read
// in 30 places in the boot segment. The game's entire libultra threading layer
// has to become the runtime's, not just the handful named so far.
osSetThreadPri = 0x8000D430; // type:func

// ---------------------------------------------------------------------------
// Message queues and thread identity. Same cause as the pair above: these read
// or write scheduler state that ultramodern now owns, so recompiling them means
// touching structures nothing maintains.
// ---------------------------------------------------------------------------

// Initialises six fields of an OSMesgQueue: mtqueue and fullqueue both point at
// the thread-tail sentinel (D_8001A330), then validCount = 0 at 0x8, first = 0
// at 0xC, msgCount = a2 at 0x10 and msg = a1 at 0x14. That field-for-field
// sequence is osCreateMesgQueue.
osCreateMesgQueue = 0x80003DC0; // type:func

// `if (t == NULL) t = __osRunningThread; return t->id;` -- the same shape as
// osGetThreadPri, reading offset 0x14 instead of 0x4. The OSThread layout is
// corroborated three ways now: priority at 0x4, state at 0x10, id at 0x14.
osGetThreadId = 0x80003EC0; // type:func

// osRecvMesg(mq, msg, flag). Reads validCount at 0x8; returns -1 immediately if
// the queue is empty and the flag is NOBLOCK; otherwise sets the running
// thread's state to OS_STATE_WAITING (8) and enqueues-and-yields, re-checking
// on wake; then copies msg[first] out through a1.
//
// This is what was crashing: `sh $s3, 0x10($v0)` writes the state field of
// __osRunningThread, which is null. The reported fault offset was 0x12 rather
// than 0x10 because MEM_H XORs the address by 2 for halfword access -- worth
// remembering when reading these, since the offset in the fault will not match
// the offset in the instruction for anything narrower than a word.
osRecvMesg = 0x800040F0; // type:func

// The osSendMesg / osJamMesg pair. Both block while validCount >= msgCount,
// both set the running thread to OS_STATE_WAITING and enqueue-and-yield, and
// both return -1 when the queue is full and the flag is NOBLOCK -- so the
// preambles are interchangeable. They are told apart by the insert index:
//
//   osSendMesg  loads first (0xC), validCount (0x8) AND msgCount (0x10) to
//               compute (first + validCount) % msgCount -- append at the tail.
//   osJamMesg   loads only first and msgCount and adds them, computing
//               (first + msgCount - 1) % msgCount -- prepend at the head. It
//               has no use for validCount in the index, and does not load it.
osSendMesg = 0x8000D2C0; // type:func
osJamMesg  = 0x8000D130; // type:func

// ---------------------------------------------------------------------------
// Video Interface.
// ---------------------------------------------------------------------------

// osCreateViManager(OSPri pri). Called from the game's main thread with
// a0 = 254, which is OS_PRIORITY_VIMGR. The body is the textbook sequence:
// a once-only guard on the manager struct at D_8001A380, timer-services init,
// osCreateMesgQueue for the VI event queue with a count of 5, the event message
// constants 0xD and 0xE, and finally creating and starting the VI thread.
//
// This is the public entry point, and taking it is the whole point. The
// function it calls to set up the two OSViContexts -- func_80011480, which
// bzeros 0x60 bytes, reads osTvType from 0x80000300 to pick NTSC/PAL/MPAL and
// sets the retrace count to 1, i.e. __osViInit -- is in the recompiler's
// *ignored* set, so naming THAT would oblige this port to emulate the VI
// registers itself. Naming osCreateViManager instead hands the entire manager
// to ultramodern, and __osViInit is simply never called.
osCreateViManager = 0x80007D10; // type:func

// ---------------------------------------------------------------------------
// Peripheral Interface (cartridge DMA).
// ---------------------------------------------------------------------------

// osCreatePiManager(pri, cmdQ, cmdBuf, cmdMsgCnt). Four arguments matching that
// signature, a once-only guard on the manager struct at D_8001B2C0, and it
// creates the caller's command queue before installing the manager's own
// routines: func_8000B6D0 as the thread entry (__osDevMgrMain -- confirmed by
// being passed as a2 to osCreateThread at 0x8000B264, and by appearing in a
// crash stack called directly from run_thread_function) and func_80013430 as
// the raw DMA routine (__osPiRawStartDma -- it writes PI_DRAM_ADDR,
// PI_CART_ADDR and PI_WR_LEN).
//
// Both of those are in the recompiler's *ignored* set, so naming either would
// oblige this port to drive the PI registers itself. Naming the public creator
// hands the whole manager -- thread, queue and DMA -- to librecomp instead.
osCreatePiManager = 0x8000B190; // type:func

// osPiStartDma -- the reason the port booted to a black screen for so long.
//
// The boot thread loads the main and aux segments by calling a wrapper
// (func_800026D8) that issues a PI DMA and then blocks in osRecvMesg waiting
// for the completion message. The probe sweep showed the thread entering that
// wrapper and never leaving it: the DMA never completed, so the main segment
// -- where the game and all of its graphics live -- was never in RAM, and
// nothing was ever submitted to the renderer.
//
// The cause is the seam between the game's libultra and the runtime's. This
// function is the real libultra osPiStartDma: it fills in an OSIoMesg and posts
// it to the PI manager's command queue, expecting the PI manager thread to pick
// it up, drive the registers and reply. But osCreatePiManager is reimplemented
// by librecomp (see above), so the game's PI manager thread does not exist. The
// message landed in a queue with no reader and the caller waited forever.
//
// osPiStartDma is in N64Recomp's reimplemented set, and elf.cpp checks that set
// before the ignored one, so naming it here replaces the body wholesale with
// librecomp's: that one performs the copy immediately and posts the completion
// message to the caller's own return queue, which is what unblocks osRecvMesg.
//
// Identified from the body rather than guessed. It writes message type 11 for
// direction 0 and 12 otherwise -- OS_MESG_TYPE_DMAREAD and DMAWRITE -- then
// fills 0x2/0x4/0x8/0xC/0x10/0x14 with priority, retQueue, dramAddr, devAddr,
// size and a NULL piHandle, which is the OSIoMesg layout field for field, and
// finally jams or sends depending on whether priority is OS_MESG_PRI_HIGH.
// librecomp's version reads dramAddr, size and the queue from sp+0x10, +0x14
// and +0x18, matching the wrapper's stack arguments exactly.
osPiStartDma = 0x8000BB80; // type:func

// osContInit -- the controller API, and the second seam of the same kind.
//
// With the PI DMA fixed the boot thread got as far as the aux segment, which
// calls this during early init. It crashed writing to 0xA4800000: the game's
// own libultra was driving the Serial Interface registers directly, and MMIO is
// not RDRAM, so the recompiler's memory macro resolved it to 0x24800000 bytes
// past the base -- inside the guard region the port maps after RDRAM, which is
// exactly why that region is mapped NOACCESS rather than left unreserved.
//
// The crash stack named the caller chain outright: func_80000854 (boot thread)
// -> func_800F69EC (aux segment init) -> this -> __osSiRawStartDma. Note that
// the return-address trick used earlier does NOT work for this: N64Recomp emits
// `jal` as a plain C call and never assigns ctx->r31, so $ra reads 0 in every
// recompiled function. The native call stack is the caller chain here.
//
// Identified from the body: it guards on a one-shot flag, creates the SI event
// message queue, registers an event handler, packs a request, issues a write
// DMA and a read DMA to the PIF with an osRecvMesg after each, unpacks the
// init data and creates the SI access queue. That is osContInit instruction
// for instruction.
//
// librecomp reimplements it against the host's input, so naming it replaces the
// whole PIF conversation with something the runtime can actually answer.
osContInit = 0x80008B10; // type:func

// The rest of the public controller API, named together.
//
// Naming osContInit alone moved the crash one function along, to the next
// routine that talks to the PIF. These three are the remainder of the surface
// the aux segment's init actually uses, and they have to be replaced as a set:
// librecomp's osContStartReadData does not fill the game's __osContPifRam
// buffer at 0x800207C0, so leaving the matching reader as game code would have
// it unpack a buffer nobody writes -- a controller that reads as permanently
// idle, which is worse than a crash because it looks like it works.
//
// osContStartReadData (0x800087F0): takes the SI access semaphore, and if the
// last command was not already a button read, packs one and issues a write DMA
// followed by a blocking osRecvMesg; then issues the read DMA, records the last
// command and releases the semaphore.
//
// osContGetReadData (0x80008878): no DMA at all -- it walks __osMaxControllers
// entries of __osContPifRam eight bytes at a time, pulls the error code out of
// bits 0xC0 shifted right by four, and stores the button halfword and the two
// stick bytes into the caller's OSContPad array.
//
// osContStartQuery (0x80008750): the same shape as StartReadData but packing a
// status request instead, and clearing the last-command byte rather than
// setting it.
//
// Along the way this also identifies, but deliberately does not name, the
// private helpers they share: func_800116D4 and func_80011740 are the SI access
// acquire/release pair, func_80008914 packs a read command and func_80008D60
// packs a request. Those are in the recompiler's ignored set, so naming any of
// them would oblige this port to implement it; replacing their public callers
// means nothing ever reaches them.
osContStartQuery    = 0x80008750; // type:func
osContStartReadData = 0x800087F0; // type:func
osContGetReadData   = 0x80008878; // type:func

// osPfsInitPak -- the Controller Pak probe, and the third seam of the same kind.
//
// Same failure signature as the controller API: a write 0x24800000 bytes past
// the RDRAM base, which is game address 0xA4800000, the SI registers. The
// crash stack read func_80000854 -> here -> func_8000A6B0 -> __osSiRawStartDma,
// so this is the routine that goes out over the SI to ask whether a pak is
// present.
//
// Identified from the structure it fills rather than from the call. It takes
// the SI access semaphore, performs a raw pak read, releases it, then writes
// the caller's struct at 0x0, 0x4 and 0x8, at 0x4C, 0x50 and 0x54, and single
// bytes at 0x64 and 0x65. That is OSPfs field for field: status, queue and
// channel at the front, version, dir_size and inode_table in the middle, and
// the bank count and active bank at the end.
//
// It is deliberately NOT named. librecomp answers PFS_ERR_NOPACK to every Pfs
// entry point, which is not a stub in the sense of being unfinished -- it is
// the answer the console gives when no Controller Pak is inserted, and a game
// that shipped had to handle it. That was the right answer while the port had
// no pak at all, and it is the wrong one now: naming this symbol replaces the
// cartridge's own PFS implementation with a refusal, and the game can never
// save.
//
// Leaving it unnamed recompiles the game's own copy, and with it the whole
// libultra filesystem underneath -- __osPfsGetStatus, __osContRamRead and
// __osContRamWrite, the bank select, the ID and inode checksums, the note
// table. That code is far better tested than anything this port could write,
// and because it is the code that shipped, the bytes it leaves in the pak image
// are a real Controller Pak image rather than an approximation of one. All the
// port has to supply is the wire underneath it: see __osSiRawStartDma below.
//
//     osPfsInitPak = 0x8000A320;   // deliberately not named

// __osSiRawStartDma -- the wire the whole Controller Pak stands on.
//
// Identified from the body: it takes a direction in $a0 and a buffer in $a1,
// faults the buffer if it is not word-aligned, spins on __osSiDeviceBusy,
// writes back or invalidates 0x40 bytes of the caller's buffer through
// osWritebackDCache / osInvalDCache, converts the buffer to a physical address
// and stores it to SI_DRAM_ADDR (0xA4800000), then stores the PIF RAM address
// 0x1FC007C0 to either SI_PIF_ADDR_RD64B (0xA4800004) for a read or
// SI_PIF_ADDR_WR64B (0xA4800010) for a write. That is __osSiRawStartDma and
// nothing else: 0x40 bytes is exactly the PIF RAM, and the two store addresses
// are the two directions of the only transfer that touches it.
//
// This name is in N64Recomp's `ignored_funcs` and NOT in `reimplemented_funcs`,
// so naming it does not hand the job to librecomp -- it drops the function and
// obliges the port to define __osSiRawStartDma_recomp. That is the point. The
// port answers the joybus transactions in src/si_pak.cpp, which is a far
// smaller surface than the filesystem above it: two commands, read block and
// write block, against a 32 KiB backing store.
//
// The game reaches it as osPfsInitPak -> func_8000A6B0 (__osPfsGetStatus) ->
// __osSiRawStartDma(OS_WRITE) -> osRecvMesg -> __osSiRawStartDma(OS_READ) ->
// osRecvMesg. Both directions have to post an SI completion or that first
// osRecvMesg never returns and the game hangs holding the SI semaphore.
__osSiRawStartDma = 0x800115C0; // type:func

// osSetIntMask -- the fourth seam, and the first one reached from real game code.
//
// The crash chain for this one runs ten frames deep and starts in the main
// segment: func_80025D30 -> func_800AD5B8 -> func_800B4D80 -> func_800B8124 ->
// func_800B8628 -> func_800B358C -> func_800B3044 -> func_80006F30 -> here.
// That is the game proper, not boot scaffolding, which is why it only became
// reachable once the section registration above was fixed.
//
// The fault was a read of 0xA430000C, MI_INTR_MASK. Identified from the body:
// it reads cop0 Status directly, folds in the complement of __OSGlobalIntMask,
// reads the MI interrupt mask to reconstruct the previous OSIntMask for the
// return value, then writes a new MI mask looked up from a table indexed by the
// requested bits before setting Status. That is osSetIntMask exactly, and no
// other libultra routine both reads and writes MI_INTR_MASK around a Status
// update.
//
// It is in the reimplemented set, so librecomp owns interrupt masking -- which
// it must, having replaced the interrupt model wholesale. Note that osGetIntMask
// is in the ignored set only, so it stays game code; nothing here calls it.
osSetIntMask = 0x8000CE00; // type:func

// The video interface, and the two routines that submit RSP tasks.
//
// With the seams above closed the game reaches its own main loop, at
// func_80026B08, and stays there -- no crash, no hang in boot, and still not a
// single display list. That is the signature of a game running correctly and
// talking to hardware nobody is listening to.
//
// All seven are in the recompiler's reimplemented set, and every one of them is
// a place where the game's own libultra would write a register or set a bit in
// a structure that only the real VI and RSP ever read.
//
// The five VI entry points were identified through the context pointer they
// share, D_8001B634, which is __osViNext. Their offsets into it are the
// __OSViContext layout: state at 0x0, framep at 0x4, modep at 0x8, control at
// 0xC and y_scale at 0x24.
//
//   osViSetMode (0x800080A0)            stores the mode pointer at 0x8, sets
//                                       state to 1 and copies the mode's
//                                       control word into 0xC.
//   osViSetSpecialFeatures (0x800080F0) walks the feature bits of its argument
//                                       and ORs matching bits into control.
//   osViSetYScale (0x80008260)          stores its float argument at 0x24 and
//                                       ORs 0x4 into state.
//   osViSwapBuffer (0x800082B0)         stores the framebuffer at 0x4 and ORs
//                                       0x10 into state. This is the one that
//                                       makes a frame appear, and as game code
//                                       it was setting a bit nothing read.
//   osViBlack (0x800085F0)              sets or clears state bit 0x20 according
//                                       to its byte argument.
//
// osSpTaskLoad (0x80007A20) is unmistakable: it calls osVirtualToPhysical seven
// times, once for each pointer field of an OSTask, writes back the data cache,
// and then moves the microcode and the task structure into IMEM at 0x04001000
// and DMEM at 0x04000FC0. osSpTaskStartGo (0x80007C2C) is the short companion
// that releases the halt.
//
// Their callers are func_8008F378, func_8008F4F4 and func_800AEB04 -- main
// segment game code, which is to say the graphics submission path itself.
osSpTaskLoad           = 0x80007A20; // type:func
osSpTaskStartGo        = 0x80007C2C; // type:func
osViSetMode            = 0x800080A0; // type:func
osViSetSpecialFeatures = 0x800080F0; // type:func
osViSetYScale          = 0x80008260; // type:func
osViSwapBuffer         = 0x800082B0; // type:func
osViBlack              = 0x800085F0; // type:func

// osViSetEvent -- where the game asks to be told about vertical retrace.
//
// Naming the VI and RSP entry points above was necessary and not sufficient:
// the game still sat in its main loop drawing nothing. This is why. A game's
// frame loop is paced by retrace -- it registers a message queue here and then
// blocks on it once per frame -- and as game code this routine only wrote the
// queue pointer into a structure in RDRAM. ultramodern's VI thread posts to the
// queue it was told about through its own osViSetEvent, was never told about
// any, and so posted nothing. The loop waited for a frame boundary that could
// never arrive, which is exactly what "no crash, no hang in boot, 60 VI fps and
// zero display lists" looks like from the outside.
//
// Identified by its three arguments landing at 0x10, 0x14 and 0x2 of
// __osViNext: the message queue, the message, and the retrace count as a
// halfword. Note the neighbouring func_80008300 also uses that context but
// reads VI_CURRENT_REG at 0xA4400010 -- that is __osViSwapContext, the interrupt
// side, which is private and stays game code because nothing will call it.
osViSetEvent = 0x80008040; // type:func

// osSetEventMesg -- the registration every other event fix was standing in for.
//
// The bisection that found this ran eight levels deep, from the game's main
// loop down to func_8009F898, which does:
//
//     osRecvMesg(&D_800F5C04, ...)        // take the controller lock
//     osContStartQuery(&D_800CF378)
//     osRecvMesg(&D_800CF378, ...)        // wait for the SI to finish
//
// and never came back from that second wait. The queue it blocks on is not the
// one it handed osContInit, so nothing the port had done so far could deliver
// to it. func_8009F478 is where it is registered:
//
//     osSetEventMesg(5 /* OS_EVENT_SI */, &D_800CF378, NULL)
//
// As game code that call only wrote a pointer into __osEventStateTab, a table
// in RDRAM that the port's runtime has no reason to read. ultramodern keeps its
// own event table and posts SI completions to whatever was registered through
// its osSetEventMesg -- which, until now, was nothing.
//
// Identified from the body: it shifts the event id left by three, adds it to a
// table at D_80025378, and stores the queue at +0 and the message at +4 with
// interrupts disabled. Eight-byte entries indexed by event id is
// __osEventStateTab, and no other routine writes it.
//
// This also supersedes the narrower fix in the fork's osContInit, which
// registered the queue passed to it so that the boot-time query had somewhere
// to land. That stays -- it is correct on its own terms and covers a game that
// never calls osSetEventMesg -- but the game does call it, later and with a
// different queue, and that registration now wins as it should.
osSetEventMesg = 0x80004220; // type:func

// The audio interface, reached once a second game thread started running.
//
// Naming osSetEventMesg unblocked the controller query, and the first thing
// that happened afterwards was a fault on a thread that had never got off the
// ground before: func_8008F5C0 -> func_800AEEF8 -> func_800AEE00 -> here. The
// address was 0xA4500004, AI_LEN.
//
// osAiGetLength (0x80007840) is four instructions and reads that register and
// nothing else. osAiSetNextBuffer (0x80007980) is the writer: it carries
// libultra's workaround for a buffer that would cross a 0x2000 boundary,
// polls the busy bit, and then stores the DRAM address and the length to
// 0xA4500000 and 0xA4500004.
//
// The busy poll itself, func_800112F0, reads AI_STATUS at 0xA450000C and tests
// the sign bit -- __osAiDeviceBusy. It is in the ignored set and stays game
// code, which is safe precisely because replacing its only caller means nothing
// reaches it.
osAiGetLength     = 0x80007840; // type:func
osAiSetNextBuffer = 0x80007980; // type:func

// osContGetQuery -- the reader half of the query, left behind by mistake.
//
// osContStartQuery was named earlier; this is its partner, and the same
// argument applies to it as applied to osContGetReadData: librecomp's
// osContStartQuery does not fill the game's __osContPifRam, so leaving the
// reader as game code has it unpack a buffer nobody writes. The game asks
// whether a controller is present and gets whatever happens to be in memory.
//
// It is four instructions -- it moves its OSContStatus argument to the second
// parameter and tail-calls __osContGetInitData (func_80008CAC) with a scratch
// byte for the bit pattern -- which is osContGetQuery exactly.
osContGetQuery = 0x800087D0; // type:func

// osViGetCurrentFramebuffer -- the last gate in front of the display list.
//
// With the busy-wait deadlock fixed the game reaches its main loop and turns it
// continuously, and every part of the graphics pipeline runs: the frame body
// builds a list, func_8008EDC0 terminates it and posts to the task thread, and
// func_8008F378 -- the routine that actually submits a graphics task -- was
// entered 524 times in fourteen seconds. It never reached its osSpTaskLoad.
//
// Its first act after taking the message is to ask which framebuffer the video
// interface is currently showing, and to skip the submission if that is already
// the buffer it was about to draw into. As game code that question is answered
// out of __osViCurr at D_8001B630, a structure in RDRAM that the port's runtime
// has no reason to maintain -- osViSwapBuffer is librecomp's now and updates
// ultramodern's own state instead. So the answer was a constant, the comparison
// matched, and the game skipped every frame it ever drew.
//
// Identified from the body: it takes the interrupt lock, loads the pointer at
// D_8001B630, reads the word at offset 4 of what it points to -- framep in
// __OSViContext, the same layout osViSwapBuffer writes -- and returns it. It is
// in the reimplemented set, so librecomp answers it from the state that is
// actually true.
osViGetCurrentFramebuffer = 0x80007CD0; // type:func

// osSpTaskYield and osSpTaskYielded -- the last raw MMIO on the graphics path.
//
// With the frame loop running the port got about seven seconds in and then
// faulted writing 0xA4040010, SP_STATUS. The chain was the task thread ->
// func_80007C60 -> func_80011340, and func_80007C60 is three instructions:
// __osSpSetStatus(0x400). SP_SET_SIG0 is what libultra's osSpTaskYield writes
// to ask the running microcode to stop, so that is what this is.
//
// Its neighbour func_80007C80 is the other half: it reads SP_STATUS, tests bit
// 0x80 -- the yielded flag -- folds bit 8 into the caller's OSTask flags and
// clears it. That is osSpTaskYielded, and it is reached from the same state
// machine, so naming one without the other would just move the fault.
//
// Both are in the reimplemented set and librecomp implements both, which it has
// to: whether a task yielded is a question about ultramodern's own task thread,
// not about a register this port does not have.
osSpTaskYield   = 0x80007C60; // type:func
osSpTaskYielded = 0x80007C80; // type:func
